\begin{frame}[t]{Data Sources and Unit of Observation}
\begin{itemize}
  \item \textbf{Healthcare Cost Report Information System (HCRIS)}
  \begin{itemize}
    \item Annual cost report information + \textbf{ownership change flag}
    \item \textbf{Initial flag} that a change may have occurred
  \end{itemize}
  \pause
  \item \textbf{Nursing Home Compare (NHC) Archive}
  \begin{itemize}
    \item Monthly ownership panel with every reported ownership layer (managing org $\rightarrow$ direct owners $\rightarrow$ indirect owners)
    \item Used to \textbf{identify} and \textbf{date} ownership changes using ownership shares
	\end{itemize}
\pause
	\item \textbf{Payroll Based Journal (PBJ)}
	\begin{itemize}
    \item Payroll-reported staffing hours by staff type
    \item Used to construct outcome variables
	\end{itemize}
\pause
  \item  facility-month panel, 48 contiguous states, Jan 2017 -- Jun 2024
\end{itemize}
\end{frame}

\begin{frame}[t]{Identifying Ownership Changes: Example (Legal Restructuring)}
\begin{itemize}
  \item HCRIS ownership change flags may reflect \textbf{purely legal restructuring}
  \item NHC ownership data allows us to verify whether economic control changes

  \vspace{0.5em}
  \item \textbf{Example from the data:}
	\begin{itemize}
    		\item \textit{Before:}
    		\begin{itemize}
      		\item ESTES JAMES (84\%)
      		\item BURCHFIELD JOHN (10\%)
      		\item LEE CLAUDE (6\%)
    		\end{itemize}

    		\item \textit{After:}
    		\begin{itemize}
     		\item JAMES N ESTES JR FAMILY DYNASTY TR NO 2 (50\%)
     		\item JENNIFER E AGEE FAMILY DYNASTY TR NO 2 (50\%)
    \end{itemize}
  \end{itemize}

  \item We do \textbf{not} treat this as a meaningful ownership change
\end{itemize}
\end{frame}

\begin{frame}[t]{Identifying Ownership Changes: Definition and Sample}
\begin{itemize}
  \item \textbf{Step 1 (flag):} HCRIS ownership change flag identifies candidate change events
  \item \textbf{Step 2 (validate \& date):} NHC ownership panel identifies whether control changes and when

  \vspace{0.5em}
  \item \textbf{Ownership change criteria (NHC):}
  \begin{itemize}
    \item An ownership change occurs between $t-1$ and $t$ when \textbf{majority control (50\%)} transfers
    \item Partial / missing shares are normalized; missing ownership shares are assigned equal weights within a level
    \item We do not flag changes driven by name changes or within-family reclassification (e.g., trust/LLC relabeling)
  \end{itemize}

  \vspace{0.5em}
  \item \textbf{Final sample restriction):}
  \begin{itemize}
    \item Facilities with \textbf{(0,0)} changes in both datasets, or \textbf{(1,1)} with dates within 6 months
    \item Change month is the first month NHC indicates a change
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Dependent Variables: Nurse Staffing (PBJ)}
\begin{itemize}
  \item We study four staffing measures:
  \begin{itemize}
    \item \textbf{Registered Nurses (RN)}: clinical assessment, medication administration, supervision
    \item \textbf{Licensed Practical Nurses (LPN)}: clinical support, medication-related tasks
    \item \textbf{Certified Nursing Assistants (CNA)}: direct daily care (bathing, feeding, mobility)
    \item \textbf{Total nursing staff}: RN + LPN + CNA
  \end{itemize}

  \vspace{0.4em}
  \item Outcome is \textbf{hours per patient day (HPPD)}: staffing intensity standardized by resident census
\end{itemize}

\vspace{0.2em}
\[
\text{HPPD}_{s,i,t}=\frac{H_{s,i,t}}{\text{PatientDays}_{i,t}}
\]
\end{frame}

\begin{frame}[t]{PBJ Cleaning and Final Sample}
\begin{itemize}
  \item We apply CMS technical guidance to remove implausible staffing reports:
  \begin{enumerate}
    \item RN HPPD = 0 \textbf{and} LPN HPPD = 0
    \item Total HPPD $< 1.5$ or $> 12$
    \item CNA HPPD $> 5.25$
  \end{enumerate}

  \vspace{0.5em}
  \item Additional restrictions:
  \begin{itemize}
    \item Drop facilities with beds $< 15$
    \item Drop hospital-based facilities
  \end{itemize}

  \vspace{0.5em}
  \item \textbf{Final sample:}
  \begin{itemize}
    \item 7,806 facilities
    \item 632,231 facility-month observations
    \item 1,589 facilities undergo a change of ownership, about 20\% of the final sample
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Empirical Strategy: Event-Study Difference-in-Differences}
\small
\begin{itemize}
  \item \textbf{Design:} staggered-adoption DiD event study around ownership transitions
  \item Identification comes from \textbf{within-facility changes} relative to facilities not yet treated
\end{itemize}

\vspace{0.2em}
\begin{equation}
Y_{it} =
\sum_{\tau=-24}^{24} \beta_{\tau}\,\mathbf{1}\{event\_time_{it}=\tau\}\cdot Treat_i
+ X'_{it}\gamma + \alpha_i + \lambda_t + \varepsilon_{it}.
\end{equation}

\vspace{0.15em}
\begin{itemize}
  \item $Y_{it}$: staffing outcome measured in \textbf{HPPD}
  \item $\alpha_i$ facility FE, $\lambda_t$ month FE; $X'_{it}$ controls (beds, occupancy, payer mix, ownership type, chain, acuity)
  \item Standard errors clustered at the facility level
\end{itemize}
\end{frame}

% =========================
% METHODS: Identification Assumption + Threat
% =========================
\begin{frame}[t]{Identification Assumption}
\begin{itemize}[<+->]
  \item \textbf{Parallel trends:} absent an ownership change, treated facilities would have followed the same staffing trends as facilities not yet treated.
  \vspace{0.4em}
  \item \textbf{Primary threat: anticipation / timing mismatch}
  \begin{itemize}
    \item Staffing may adjust before the recorded transition month.
  \end{itemize}
\end{itemize}
\end{frame}

% =========================
% METHODS: Motivation for Donut (put BEFORE donut)
% =========================
\begin{frame}[t]{Why Anticipation is Plausible in Ownership Transitions}
\begin{itemize}[<+->]
  \item Ownership changes are not instantaneous: negotiations and operational handoffs occur before the recorded change month
  \item Staffing can adjust during transition planning (turnover, hiring freezes, contract renegotiations)
  \item Administrative “event dates” may reflect reporting/legal dates, not the first operational change
  \item \textbf{Implication:} months immediately before $\tau=0$ may violate parallel trends
\end{itemize}
\end{frame}

% =========================
% METHODS: Donut Specification (now it feels motivated)
% =========================
\begin{frame}[t]{Main Specification: Donut Event Study}
\begin{itemize}[<+->]
  \item To handle this, we exclude months $\tau \in \{-3,-2,-1\}$
  \item We use $\tau=-4$ as the reference period
  \item Post change coefficients compare to a pre-period less likely to be influenced by transition activity
\end{itemize}
\end{frame}

% =========================
% METHODS: Parallel Trends Assessment (setup)
% =========================
\begin{frame}[t]{Assessing Parallel Trends}
\begin{itemize}
  \item We test parallel trends using a joint Wald test of
  \[
    H_0:\ \beta_{\tau}=0\ \ \forall\ \tau<0.
  \]
  \vspace{0.3em}
  \pause
  \item \textbf{Takeaway:}
  \begin{itemize}
    \item With the full pre-window (including $\tau=-3,-2,-1$), pre-trends show evidence of violation.
    \item Excluding anticipatory months (donut) yields pre-trends more consistent with parallel trends.
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Summary Effect: TWFE Difference-in-Differences}
\small
\[
Y_{it}=\beta\,Post_{it} + X'_{it}\gamma + \alpha_i + \lambda_t + \varepsilon_{it}
\]
\begin{itemize}
	\item $Post_{it} = 0$ before an ownership change and $=1$ in all months after a change
  \item $\beta$ summarizes the average post-change staffing effect
  \item Quantifies average changes for cleaner and relevant results
\end{itemize}
\end{frame}
