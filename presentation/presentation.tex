% INTRODUCTION + BACKGROUND (RQ EARLIER)

\begin{frame}[t]{Motivation: Ownership Changes Beyond Market Structure}
\begin{itemize}%[<+->]
  \item Ownership transitions are common in healthcare.
  \item IO research often emphasizes \textbf{prices, market power, and concentration}.
  \item But many transitions do \textbf{not} change the set of providers in a market.
  \item In those cases, effects operate through \textbf{internal production decisions}:
  management practices and especially \textbf{staffing inputs}.
\end{itemize}
\end{frame}

\begin{frame}[t]{Research Question}
\begin{itemize}%[<+->]
  \item \textbf{Research question:} How do nurse staffing levels respond to ownership changes in nursing homes?
  \begin{itemize}
  \item Do responses differ across staff type or organizational structure?
  \item Do responses differ pre vs post-pandemic?
  \end{itemize}
  \item Why we care: staffing is the primary input into care, so ownership changes can affect \textbf{quality} even when market structure is unchanged.
\end{itemize}
\end{frame}

\begin{frame}[t]{Why Nursing Homes?}
\begin{itemize}%[<+->]
  \item Nursing home care is labor-intensive and heavily regulated.
  \item \textbf{Nursing labor is the main input} into care and a major operating cost.
  \item Staffing is a natural adjustment margin after ownership change, but it’s constrained by:
  \begin{itemize}
    \item compliance,
    \item resident outcomes,
    \item demand/occupancy
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Nursing Staff Types}
\begin{itemize}%[<+->]
  \item We study four staffing measures:
  \begin{itemize}
    \item \textbf{RN (Registered Nurses):} clinical assessment, care planning, meds, supervision
    \item \textbf{LPN (Licensed Practical Nurses):} routine clinical tasks, meds under RN oversight
    \item \textbf{CNA (Certified Nursing Assistants):} direct daily care
    \item \textbf{Total nursing:} RN + LPN + CNA
  \end{itemize}
  \item Ownership change can affect \textbf{levels} (total hours) and \textbf{mix} (skill composition).
\end{itemize}
\end{frame}

\begin{frame}[t]{The Cost--Quality Tradeoff in Staffing}
\begin{itemize}%[<+->]
  \item Owners can raise quality through:
  \begin{itemize}
    \item \textbf{Quantity} of staffing and/or
    \item \textbf{Quality / skill mix}
  \end{itemize}
  \item Either margin increases costs
  \item But \textbf{revenues are relatively flat in the short run}
  \item So owners face a tradeoff: \textbf{higher quality vs. tighter margins}.
\end{itemize}
\end{frame}

\begin{frame}[t]{Why Staffing Could Rise or Fall After a Transition}
\begin{itemize}%[<+->]
  \item \textbf{Cost pressures:} tighter budgets / debt service can push cost cutting.
  \item \textbf{Reorganization:} standardized policies may shift RN/LPN/CNA composition.
  \item \textbf{Quality objective:} owners with stronger quality/compliance goals may increase staffing.
  \item Net effect is ambiguous $\Rightarrow$ \textbf{empirical question}.
\end{itemize}
\end{frame}

\begin{frame}[t]{Data Sources and Unit of Observation}
\begin{itemize}
  \item \textbf{Healthcare Cost Report Information System (HCRIS)}
  \begin{itemize}
    \item Annual cost report information
    \item Contains information about costs, revenue, financial data and dates of ownership change
  \end{itemize}
  \item \textbf{Nursing Home Compare (NHC) Archive}
  \begin{itemize}
    \item Contains ownership information as well as information on chain affiliation
	\end{itemize}
	\item \textbf{Payroll Based Journal (PBJ)}
	\begin{itemize}
    \item Payroll-reported staffing hours by staff type
    \item Used to construct outcome variables
	\end{itemize}

  \item \textbf{Facility-month panel, 48 contiguous states, Jan 2017 -- Jun 2024}
\end{itemize}
\end{frame}

\begin{frame}[t]{Identifying Ownership Changes: Step 1}
\small
\begin{itemize}
  \item \textbf{Starting point (HCRIS):} includes a variable indicating \textbf{whether and when} an ownership change occurred.
  \vspace{0.35em}
  \item \textbf{Challenge:} ownership is often \textbf{layered} (organizations owning organizations), so an administrative flag may not map cleanly to an \textbf{economic change in control}.
  \vspace{0.35em}
  \item \textbf{Check:} NHC archive provides owner identities and structure:
  \begin{itemize}
    \item \textbf{Owner name}, \textbf{ownership level}, and \textbf{owner type}
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Identifying Ownership Changes: Step 2}
\small
\begin{itemize}
  \item \textbf{Validate \& date using NHC:}
  \begin{itemize}
    \item Track ownership shares over time within the highest reported ownership level
    \item Define a change when \textbf{majority control (50\%)} transfers between $t-1$ and $t$
    \item We do not flag changes driven by name changes or within-family reclassification
  \end{itemize}

  \vspace{0.35em}

  \item \textbf{Example from the data:}
  \begin{itemize}
    \item \textit{Before:}
    \begin{itemize}
      \item ESTES JAMES (84\%)
      \item BURCHFIELD JOHN (10\%)
      \item LEE CLAUDE (6\%)
    \end{itemize}

    \item \textit{After:}
    \begin{itemize}
      \item JAMES N ESTES JR FAMILY DYNASTY TR NO 2 (50\%)
      \item JENNIFER E AGEE FAMILY DYNASTY TR NO 2 (50\%)
    \end{itemize}

    \item We do \textbf{not} treat this as a meaningful ownership change
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Dependent Variables: Nurse Staffing (PBJ)}
\begin{itemize}
  \item We study four staffing measures:
  \begin{itemize}
    \item \textbf{Registered Nurses (RN)}: clinical assessment, medication administration, supervision
    \item \textbf{Licensed Practical Nurses (LPN)}: clinical support, medication-related tasks
    \item \textbf{Certified Nursing Assistants (CNA)}: direct daily care (bathing, feeding, mobility)
    \item \textbf{Total nursing staff}: RN + LPN + CNA
  \end{itemize}

  \vspace{0.4em}
  \item Outcome is \textbf{hours per patient day (HPPD)}: staffing intensity standardized by resident census
\end{itemize}

\vspace{0.2em}
\[
\text{HPPD}_{s,i,t}=\frac{H_{s,i,t}}{\text{PatientDays}_{i,t}}
\]
\end{frame}

\begin{frame}[t]{Final Sample Construction}
\small
\begin{itemize}
  \item \textbf{Validated ownership-change sample (HCRIS $\leftrightarrow$ NHC):}
  \begin{itemize}
    \item Keep facilities with \textbf{(0,0)} changes in both datasets, or \textbf{(1,1)} with dates within 6 months
    \item Event month is the \textbf{first month} NHC indicates a change
  \end{itemize}

  \vspace{0.35em}
  \item \textbf{CMS implausible staffing}
  \begin{itemize}
    \item RN HPPD $=0$ \textbf{and} LPN HPPD $=0$
    \item Total HPPD $<1.5$ or $>12$
    \item CNA HPPD $>5.25$
  \end{itemize}

  \vspace{0.35em}
  \item \textbf{Additional restrictions:} beds $<15$; hospital-based facilities

  \vspace{0.35em}
  \item \textbf{Final sample:} 7,806 facilities; 632,231 facility-months; 1,589 are treated, about 2\% of the final sample
\end{itemize}
\end{frame}

\begin{frame}[t]{Empirical Strategy: Event-Study Difference-in-Differences}
\small
\begin{itemize}
  \item \textbf{Design:} staggered-adoption DiD around ownership transitions
  \item \textbf{Variation:} within-facility staffing changes relative to facilities \textit{not yet treated}
  \vspace{0.35em}
  \item \textbf{Event study (dynamic effects):}
  \[
  Y_{it} =
  \sum_{\tau=-24}^{24} \beta_{\tau}\,\mathbf{1}\{event\_time_{it}=\tau\}\cdot Treat_i
  + X'_{it}\gamma + \alpha_i + \lambda_t + \varepsilon_{it}.
  \]
  \vspace{0.2em}
  \item \textbf{Key objects:}
  \begin{itemize}
    \item $Y_{it}$: staffing outcome in \textbf{HPPD} (RN, LPN, CNA, Total)
    \item $\beta_{\tau}$: effect at event time $\tau$ (relative to the reference period)
    \item $\alpha_i$ facility FE; $\lambda_t$ month FE; $X_{it}$ controls (beds, occupancy, payer mix, ownership type, chain, acuity)
    \item SEs clustered at the facility level
  \end{itemize}
\end{itemize}
\end{frame}

% =========================================================
% Slide 2: TWFE DiD
% =========================================================
\begin{frame}[t]{Empirical Strategy: TWFE Difference-in-Differences}
\small
\begin{itemize}
  \item \textbf{Goal:} summarize the \textbf{average} staffing change after an ownership transition
  \vspace{0.35em}
  \item \textbf{TWFE DiD (headline effect):}
  \[
  Y_{it}=\beta\,Post_{it} + X'_{it}\gamma + \alpha_i + \lambda_t + \varepsilon_{it}
  \]
  \vspace{0.2em}
  \item \textbf{Key objects:}
  \begin{itemize}
    \item $Post_{it}=1$ in all months after a change ($=0$ before)
    \item $\beta$: \textbf{average post-change effect} on staffing (HPPD)
    \item Same fixed effects and controls as the event-study specification
    \item SEs clustered at the facility level
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Identification: Parallel Trends}
\small
\begin{itemize}
  \item \textbf{Parallel trends:} absent an ownership change, treated facilities would have followed the same staffing trends as facilities not yet treated.
  \vspace{0.4em}
  \item \textbf{Why anticipation is plausible in ownership transitions}
  \begin{itemize}
    \item Changes are not instantaneous: negotiations and operational handoffs precede the recorded month
    \item Staffing can adjust during planning (turnover, hiring freezes, renegotiated contracts)
    \item Recorded dates may reflect administrative/legal timing, not first operational change
  \end{itemize}
  \vspace{0.35em}
  \item \textbf{Implication:} months just before $\tau=0$ may violate parallel trends.
\end{itemize}
\end{frame}

\begin{frame}[t]{Main Specification: Donut Event Study + Pre-trend Test}
\small
\begin{itemize}
  \item \textbf{Donut event study:} mitigate this influence by excluding months just before the recorded transition.
  \vspace{0.35em}
  \item \textbf{Implementation}
  \begin{itemize}
    \item Exclude $\tau \in \{-3,-2,-1\}$
    \item Use $\tau=-4$ as the reference period
  \end{itemize}
  \vspace{0.35em}
  \item \textbf{Parallel Trends:} joint Wald test of pre-period leads
  \[
    H_0:\ \beta_{\tau}=0 \quad \forall\ \tau<0
  \]
  \vspace{0.2em}
  \item \textbf{Takeaway:} full pre-window shows evidence of pre-trends; the donut specification is more consistent with parallel trends.
\end{itemize}
\end{frame}
