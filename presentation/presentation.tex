\begin{frame}[t]{Identifying Ownership Changes - Criteria}
\begin{enumerate}
	\item Majority (50\%) of ownership transfers from month $t-1$ to month $t$
	\item Partial percentages we normalize and missing we assign equal weights across
	\item We don't flag ownership changes when 80\% stake remains in the same family or for 		changes in legal names (e.g., a change from “Smith Holdings LLC” to “Smith Family 				Partners LP”)
	\item Our sample then included facilities that facilities that:
	\begin{enumerate}
		\item Have no ownership changes in either dataset
		\item Have exactly one change in each dataset within six months of each other
	\end{enumerate}
	\item For these facilities we define the change month as the first month the NHC archive 	indicates a change
\end{enumerate}
\end{frame}

\begin{frame}[t]{Dependent Variables - Nurse Staffing Levels}
\begin{itemize}
	\item The staff types we look at are RN's, LPN's, CNA's, and total staffing
	\item PBJ
	\item Hours per patient day (HPPD) for each staff type $s$, facility $i$, and period $t$
\end{itemize}

\begin{equation}
\textcolor{red}{
  \text{HPPD}_{i,t}
  =
  \frac{H_{s,i,t}}{R_{s,i,t}}
}
\end{equation}


where $H_{s,i,t}$ is the total hours worked by the staff type $s$ during month $t$ for facility $i$ and $R_{s,i,t}$ is the total resident days in that facility month 
\end{frame}

\begin{frame}[t]{Dependent Variables - Nurse Staffing Levels}
\begin{itemize}
	\item We exclude a small number of observations to align with the implausible criteria 		outlined in the CMS NHC Technical Users Guide where:
	\begin{enumerate}
		\item RN and LPN HPPD are both 0 or
		\item Total HPPD is below 1.5 or above 12 or
		\item CNA HPPD exceeds 5.25
	\end{enumerate}
\end{itemize}
\end{frame}

  \begin{itemize}
    \item Monthly ownership roster with \textit{every reported ownership layer} (managing org $\rightarrow$ direct owners $\rightarrow$ indirect owners)

  \begin{itemize}
    \item Payroll-reported staffing hours by staff type
    \item Used to construct \textbf{hours per patient day (HPPD)} outcomes
  \end{itemize}

  \item \textbf{Sample frame:} facility-month panel, 48 contiguous states, Jan 2017 -- Jun 2024
\end{itemize}
\end{frame}

\begin{frame}[t]{Identifying Ownership Changes: Example (Legal Restructuring)}
\begin{itemize}
  \item HCRIS ownership change flags may reflect \textbf{purely legal restructuring} (not a real transfer of control)
  \item NHC ownership rosters allow us to verify whether economic control changes

  \vspace{0.5em}
  \item \textbf{Example from the data (same family control):}
  \begin{itemize}
    \item \textit{Before:} ESTES JAMES (84\%); BURCHFIELD JOHN (10\%); LEE CLAUDE (6\%)
    \item \textit{After:} JAMES N ESTES JR FAMILY DYNASTY TR NO 2 (50\%); JENNIFER E AGEE FAMILY DYNASTY TR NO 2 (50\%)
  \end{itemize}

  \item Interpretation: restructuring into trusts/vehicles, but \textbf{control remains within the same family}
  \item We do \textbf{not} treat this as a meaningful ownership change
\end{itemize}
\end{frame}

\begin{frame}[t]{Identifying Ownership Changes: Definition and Sample}
\begin{itemize}
  \item \textbf{Step 1 (flag):} HCRIS ownership change flag identifies candidate change events
  \item \textbf{Step 2 (validate \& date):} NHC ownership rosters identify whether control changes and when

  \vspace{0.5em}
  \item \textbf{Ownership change definition (NHC):}
  \begin{itemize}
    \item An ownership change occurs between $t-1$ and $t$ when \textbf{majority control (50\%)} transfers
    \item Partial / missing shares are normalized; missing ownership shares are assigned equal weights within a level
    \item We do not flag changes driven by name changes or within-family reclassification (e.g., trust/LLC relabeling)
  \end{itemize}

  \vspace{0.5em}
  \item \textbf{Final sample restriction (agreement):}
  \begin{itemize}
    \item Facilities with \textbf{(0,0)} changes in both datasets, or \textbf{(1,1)} with dates within 6 months
    \item Change month is the first month NHC indicates a change
  \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[t]{Dependent Variables: Nurse Staffing (PBJ)}
\begin{itemize}
  \item PBJ reports payroll-based staffing hours by staff category
  \item We study four staffing measures:
  \begin{itemize}
    \item \textbf{Registered Nurses (RN)}: clinical assessment, medication administration, supervision
    \item \textbf{Licensed Practical Nurses (LPN)}: clinical support, medication-related tasks (scope varies by state)
    \item \textbf{Certified Nursing Assistants (CNA)}: direct daily care (bathing, feeding, mobility)
    \item \textbf{Total nursing staff}: RN + LPN + CNA
  \end{itemize}

  \vspace{0.4em}
  \item Outcome is \textbf{hours per patient day (HPPD)}: staffing intensity standardized by resident census
\end{itemize}

\vspace{0.2em}
\[
\text{HPPD}_{s,i,t}=\frac{H_{s,i,t}}{\text{ResidentDays}_{i,t}}
\]
\end{frame}

\begin{frame}[t]{PBJ Cleaning and Final Sample}
\begin{itemize}
  \item We apply CMS technical guidance to remove implausible staffing reports:
  \begin{enumerate}
    \item RN HPPD = 0 \textbf{and} LPN HPPD = 0
    \item Total HPPD $< 1.5$ or $> 12$
    \item CNA HPPD $> 5.25$
  \end{enumerate}

  \vspace{0.5em}
  \item Additional restrictions:
  \begin{itemize}
    \item Drop facilities with beds $< 15$
    \item Drop hospital-based facilities
  \end{itemize}

  \vspace{0.5em}
  \item \textbf{Final sample:}
  \begin{itemize}
    \item 7,806 facilities
    \item 632,231 facility-month observations
  \end{itemize}
\end{itemize}
\end{frame}
