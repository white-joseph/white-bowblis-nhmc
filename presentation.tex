\begin{frame}
\frametitle{Event Study}
\begin{equation}
\label{eq:es_twfe}
Y_{it}
=
\sum_{\tau = -24}^{24}
    \beta_{\tau}\,
    \mathbf{1}\!\left\{ \text{event\_time}_{it} = \tau \right\}
    \cdot \text{Treat}_{i}
+
X_{it}^{\prime}\gamma
+
\alpha_i
+
\lambda_t
+
\varepsilon_{it}.
\end{equation}

\begin{itemize}
\item $Y_{i,t}$ denotes nurse HPPD for facility $i$ in month $t$
\item $\text{event\_time}_{it}$ measures the number of months relative to the month of ownership change
\item $\text{Treat}_{i,t}$ is a dummy for if a facility is treated or not
\end{itemize}

and our variable of interest is $\beta_{\tau}$ which captures the effect of an ownership change in month $\tau$ after the event
\end{frame}
